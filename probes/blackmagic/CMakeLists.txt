cmake_minimum_required(VERSION 3.21)
get_filename_component(PROGRAMMER_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${PROGRAMMER_ROOT}/pico_sdk_import.cmake")
set(MIOLINK_SOURCE "${PROGRAMMER_ROOT}/.tools/miolink/firmware" CACHE PATH "Pinned MioLink sources")
set(FREERTOS_KERNEL_PATH "${MIOLINK_SOURCE}/thirdParty/FreeRTOS-Kernel")
include("${MIOLINK_SOURCE}/FreeRTOS_Kernel_import.cmake")
project(universal_blackmagic C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(TINYUSB_OPT_OS OPT_OS_FREERTOS)
set(PICO_TINYUSB_PATH "${MIOLINK_SOURCE}/thirdParty/tinyusb")
pico_sdk_init()
if(NOT PICO_BOARD STREQUAL "pico" OR NOT PICO_SDK_VERSION_STRING VERSION_EQUAL "2.2.0")
    message(FATAL_ERROR "Use original Pico with pinned SDK 2.2.0")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/integration")
execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_LIST_DIR}/prepare.py"
    "${MIOLINK_SOURCE}" "${GENERATED}" COMMAND_ERROR_IS_FATAL ANY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/prepare.py")
add_executable(blackmagic board.c "${GENERATED}/main.c" "${GENERATED}/platform.c" "${GENERATED}/gdb_if.c")
set(BMP "${MIOLINK_SOURCE}/thirdParty/blackmagic")
foreach(SOURCE command exception gdb_main gdb_packet hex_utils maths_utils morse remote rtt timing)
    target_sources(blackmagic PRIVATE "${BMP}/src/${SOURCE}.c")
endforeach()
file(GLOB TARGET_SOURCES CONFIGURE_DEPENDS "${BMP}/src/target/*.c")
list(FILTER TARGET_SOURCES EXCLUDE REGEX ".*/(adiv5_swd|swdptap_generic)\\.c$")
target_sources(blackmagic PRIVATE ${TARGET_SOURCES})
file(GLOB PLATFORM_SOURCES CONFIGURE_DEPENDS "${MIOLINK_SOURCE}/source/bmp/*.c")
list(FILTER PLATFORM_SOURCES EXCLUDE REGEX ".*/(platform|platform_boards|gdb_if)\\.c$")
target_sources(blackmagic PRIVATE ${PLATFORM_SOURCES})
target_include_directories(blackmagic BEFORE PRIVATE "${GENERATED}")
target_include_directories(blackmagic PRIVATE "${BMP}/src/include" "${BMP}/src/target"
    "${MIOLINK_SOURCE}/include" "${MIOLINK_SOURCE}/include/bmp")
target_compile_definitions(blackmagic PRIVATE PICO_FLASH_SIZE_BYTES=2097152
    PICO_STDIO_SHORT_CIRCUIT_CLIB_FUNCS=0 ENABLE_DEBUG=0 ENABLE_SYSVIEW_TRACE=0
    GDB_PACKET_BUFFER_SIZE=8192 PC_HOSTED=0 NO_LIBOPENCM3 DFU_SERIAL_LENGTH=17
    SYSTICKHZ=100 ENABLE_RTT=1 PLATFORM_IDENT_DYNAMIC)
target_compile_options(blackmagic PRIVATE -Os -UNDEBUG)
pico_set_printf_implementation(blackmagic compiler)
pico_enable_stdio_usb(blackmagic 0)
pico_enable_stdio_uart(blackmagic 0)
target_link_libraries(blackmagic hardware_pio hardware_uart hardware_dma hardware_timer hardware_adc
    pico_stdlib pico_unique_id pico_bootrom pico_multicore tinyusb_device tinyusb_board
    FreeRTOS-Kernel FreeRTOS-Kernel-Heap4)
foreach(PIO pio_swd_miolink_rev_a pio_swd_miolink_rev_b pio_swd_miolink_pico pio_swd_pico pio_jtag pio_traceswo_manchester)
    pico_generate_pio_header(blackmagic "${MIOLINK_SOURCE}/source/bmp/pio/${PIO}.pio")
endforeach()
include("${PROGRAMMER_ROOT}/cmake/ProbeSlot.cmake")
programmer_probe_slot(blackmagic blackmagic default)
pico_add_extra_outputs(blackmagic)
