cmake_minimum_required(VERSION 3.21)
get_filename_component(PROGRAMMER_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${PROGRAMMER_ROOT}/pico_sdk_import.cmake")
set(DEBUGPROBE_SOURCE "${PROGRAMMER_ROOT}/.tools/debugprobe" CACHE PATH "Pinned debugprobe sources")
set(FREERTOS_KERNEL_PATH "${DEBUGPROBE_SOURCE}/freertos")
include("${DEBUGPROBE_SOURCE}/FreeRTOS_Kernel_import.cmake")
project(universal_debugprobe C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
if(NOT PICO_BOARD STREQUAL "pico" OR NOT PICO_SDK_VERSION_STRING VERSION_EQUAL "2.2.0")
    message(FATAL_ERROR "Use original Pico with pinned SDK 2.2.0")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/integration")
execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_LIST_DIR}/prepare.py"
    "${DEBUGPROBE_SOURCE}" "${GENERATED}" COMMAND_ERROR_IS_FATAL ANY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/prepare.py")
add_executable(debugprobe "${GENERATED}/main.c" swj_sequence.c dap_guard.c pin_control.c
    "${GENERATED}/tusb_edpt_handler.c" "${GENERATED}/sw_dp_pio.c" "${GENERATED}/probe.c")
foreach(SOURCE probe_config usb_descriptors cdc_uart get_serial autobaud)
    target_sources(debugprobe PRIVATE "${DEBUGPROBE_SOURCE}/src/${SOURCE}.c")
endforeach()
set_source_files_properties("${GENERATED}/sw_dp_pio.c"
    PROPERTIES COMPILE_DEFINITIONS "SWJ_Sequence=pio_swj_sequence")
foreach(SOURCE DAP JTAG_DP DAP_vendor SWO)
    target_sources(debugprobe PRIVATE "${DEBUGPROBE_SOURCE}/CMSIS_DAP/CMSIS/DAP/Firmware/Source/${SOURCE}.c")
endforeach()
target_include_directories(debugprobe BEFORE PRIVATE "${GENERATED}" "${CMAKE_CURRENT_LIST_DIR}")
target_include_directories(debugprobe PRIVATE "${DEBUGPROBE_SOURCE}/include" "${DEBUGPROBE_SOURCE}/src"
    "${DEBUGPROBE_SOURCE}/CMSIS_DAP/CMSIS/DAP/Firmware/Include"
    "${DEBUGPROBE_SOURCE}/CMSIS_DAP/CMSIS/Core/Include")
foreach(PIO probe probe_oen autobaud)
    pico_generate_pio_header(debugprobe "${DEBUGPROBE_SOURCE}/src/${PIO}.pio")
endforeach()
target_compile_definitions(debugprobe PRIVATE DEBUG_ON_PICO=1 PICO_RP2040_USB_DEVICE_ENUMERATION_FIX=1)
target_compile_options(debugprobe PRIVATE -Wall -Wextra)
pico_enable_stdio_usb(debugprobe 0)
pico_enable_stdio_uart(debugprobe 0)
target_link_libraries(debugprobe pico_multicore pico_stdlib pico_unique_id tinyusb_device tinyusb_board
    hardware_pio hardware_dma hardware_irq hardware_clocks FreeRTOS-Kernel FreeRTOS-Kernel-Heap1)
include("${PROGRAMMER_ROOT}/cmake/ProbeSlot.cmake")
programmer_probe_slot(debugprobe cmsis_dap copy_to_ram)
pico_add_extra_outputs(debugprobe)
